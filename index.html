Cara Menyambungkan ke Google Spreadsheets (2 Menit):

  1. Buka Google Sheets baru, beri nama misal: Rekap Kuis Seni Budaya 7.

  3. Hapus semua kode yang ada, lalu tempel (paste) kode berikut ini:
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Waktu Kirim", "Nama Siswa", "Kelas", "No Absen", "Skor Akhir", "Benar", "Salah", "Pelanggaran", "Status Ujian", "Durasi"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date().toLocaleString("id-ID"),
    data.nama,
    data.kelas,
    data.absen,
    data.skor,
    data.benar,
    data.salah,
    data.pelanggaran || "0x Pindah Tab",
    data.statusUjian || "Normal",
    data.durasi
  ]);
  return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
